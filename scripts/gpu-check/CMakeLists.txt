cmake_minimum_required(VERSION 3.10)
project(drone_gpu_check CXX)
set(CMAKE_CXX_STANDARD 14)
find_package(OpenCV 3.3 REQUIRED COMPONENTS core imgproc cudaimgproc cudaoptflow)
find_package(Ceres 2.1 REQUIRED)
add_executable(drone_gpu_check ../gpu_check.cpp)
target_include_directories(drone_gpu_check BEFORE PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(drone_gpu_check PRIVATE ${OpenCV_LIBS} Ceres::ceres)
set_target_properties(drone_gpu_check PROPERTIES NO_SYSTEM_FROM_IMPORTED ON INSTALL_RPATH "${CMAKE_INSTALL_PREFIX}/opencv/lib;${CMAKE_INSTALL_PREFIX}/ceres/lib;/usr/local/cuda-10.2/lib64")
install(TARGETS drone_gpu_check RUNTIME DESTINATION bin)
